## 1. Enhanced API Client with RAG Integration

```typescript
// src/lib/server/api-client.enhanced.ts
import { redis } from './redis';
import type { Document, Case, SearchResult } from './types';

interface RAGIngestionRequest {
  fileId: string;
  caseId: string;
  userId: string;
  processingOptions: {
    generateSummary: boolean;
    extractCitations: boolean;
    performEntityExtraction: boolean;
    enableSemanticSearch: boolean;
  };
}

interface RAGIngestionResponse {
  status: 'queued' | 'processing' | 'completed' | 'failed';
  jobId: string;
  estimatedCompletion: string;
  stages: Array<{
    name: string;
    status: 'pending' | 'running' | 'completed' | 'failed';
    progress: number;
  }>;
}

export class EnhancedApiClient {
  private baseUrl: string;
  private authToken?: string;

  constructor(baseUrl = process.env.MICROSERVICES_BASE_URL || 'http://localhost:8000') {
    this.baseUrl = baseUrl;
  }

  setAuthToken(token: string) {
    this.authToken = token;
  }

  private async makeRequest<T>(endpoint: string, options: RequestInit = {}): Promise<T> {
    const url = `${this.baseUrl}${endpoint}`;
    const headers = {
      'Content-Type': 'application/json',
      'Authorization': this.authToken ? `Bearer ${this.authToken}` : '',
      ...options.headers,
    };

    const response = await fetch(url, {
      ...options,
      headers,
    });

    if (!response.ok) {
      throw new Error(`API request failed: ${response.status} ${response.statusText}`);
    }

    return response.json();
  }

  // Enhanced RAG Service Integration
  async triggerRAGIngestion(request: RAGIngestionRequest): Promise<RAGIngestionResponse> {
    const cacheKey = `rag_ingestion:${request.fileId}`;
    
    // Check if ingestion is already in progress
    const cached = await redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    const response = await this.makeRequest<RAGIngestionResponse>('/rag/ingest', {
      method: 'POST',
      body: JSON.stringify(request),
    });

    // Cache the response for 5 minutes
    await redis.setex(cacheKey, 300, JSON.stringify(response));
    
    return response;
  }

  async getRAGJobStatus(jobId: string): Promise<RAGIngestionResponse> {
    const cacheKey = `rag_job:${jobId}`;
    
    // Try cache first
    const cached = await redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    const response = await this.makeRequest<RAGIngestionResponse>(`/rag/status/${jobId}`);
    
    // Cache completed jobs for 1 hour, others for 30 seconds
    const ttl = response.status === 'completed' ? 3600 : 30;
    await redis.setex(cacheKey, ttl, JSON.stringify(response));
    
    return response;
  }

  // File Upload with Progress
  async uploadFile(file: File, caseId: string, onProgress?: (progress: number) => void): Promise<{
    fileId: string;
    uploadUrl: string;
  }> {
    // Get presigned upload URL from storage service
    const { uploadUrl, fileId } = await this.makeRequest<{ uploadUrl: string; fileId: string }>('/storage/presign', {
      method: 'POST',
      body: JSON.stringify({
        filename: file.name,
        contentType: file.type,
        size: file.size,
        caseId,
      }),
    });

    // Upload file with progress tracking
    return new Promise((resolve, reject) => {
      const xhr = new XMLHttpRequest();
      
      xhr.upload.addEventListener('progress', (event) => {
        if (event.lengthComputable && onProgress) {
          const progress = Math.round((event.loaded / event.total) * 100);
          onProgress(progress);
        }
      });

      xhr.addEventListener('load', () => {
        if (xhr.status === 200) {
          resolve({ fileId, uploadUrl });
        } else {
          reject(new Error(`Upload failed: ${xhr.status}`));
        }
      });

      xhr.addEventListener('error', () => {
        reject(new Error('Upload failed'));
      });

      xhr.open('PUT', uploadUrl);
      xhr.setRequestHeader('Content-Type', file.type);
      xhr.send(file);
    });
  }

  // Hybrid Search with Caching
  async searchDocuments(query: string, caseId: string, filters: {
    documentTypes?: string[];
    dateRange?: { start: string; end: string };
    tags?: string[];
  } = {}): Promise<SearchResult[]> {
    const cacheKey = `search:${caseId}:${btoa(query)}:${JSON.stringify(filters)}`;
    
    // Check cache
    const cached = await redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    const results = await this.makeRequest<SearchResult[]>('/search/hybrid', {
      method: 'POST',
      body: JSON.stringify({ query, caseId, filters }),
    });

    // Cache search results for 5 minutes
    await redis.setex(cacheKey, 300, JSON.stringify(results));
    
    return results;
  }

  // Case Management
  async getCases(userId: string): Promise<Case[]> {
    const cacheKey = `user_cases:${userId}`;
    
    const cached = await redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    const cases = await this.makeRequest<Case[]>(`/cases?userId=${userId}`);
    
    await redis.setex(cacheKey, 300, JSON.stringify(cases));
    return cases;
  }

  async createCase(caseData: Partial<Case>): Promise<Case> {
    const newCase = await this.makeRequest<Case>('/cases', {
      method: 'POST',
      body: JSON.stringify(caseData),
    });

    // Invalidate user's cases cache
    if (caseData.userId) {
      await redis.del(`user_cases:${caseData.userId}`);
    }

    return newCase;
  }

  // Report Generation
  async generateReport(caseId: string, options: {
    includeAllDocuments?: boolean;
    documentIds?: string[];
    template?: string;
    format?: 'pdf' | 'docx' | 'html';
  }): Promise<{ reportId: string; downloadUrl: string }> {
    return this.makeRequest('/reports/generate', {
      method: 'POST',
      body: JSON.stringify({ caseId, ...options }),
    });
  }
}

export const apiClient = new EnhancedApiClient();
```

## 2. SvelteKit API Routes with Active Caching

```typescript
// src/routes/api/evidence/upload/+server.ts
import type { RequestHandler } from './$types';
import { json, error } from '@sveltejs/kit';
import { apiClient } from '$lib/server/api-client.enhanced';
import { validateSession } from '$lib/server/auth';
import { redis } from '$lib/server/redis';

export const POST: RequestHandler = async ({ request, cookies }) => {
  // Validate session
  const session = await validateSession(cookies);
  if (!session) {
    throw error(401, 'Unauthorized');
  }

  const formData = await request.formData();
  const file = formData.get('file') as File;
  const caseId = formData.get('caseId') as string;
  const processingOptions = JSON.parse(formData.get('options') as string || '{}');

  if (!file || !caseId) {
    throw error(400, 'Missing required fields');
  }

  try {
    // Step 1: Upload file to storage service
    const { fileId } = await apiClient.uploadFile(file, caseId);

    // Step 2: Trigger RAG ingestion pipeline
    const ragResponse = await apiClient.triggerRAGIngestion({
      fileId,
      caseId,
      userId: session.userId,
      processingOptions: {
        generateSummary: true,
        extractCitations: true,
        performEntityExtraction: true,
        enableSemanticSearch: true,
        ...processingOptions,
      },
    });

    // Step 3: Invalidate related caches
    await redis.del(`case_documents:${caseId}`);
    await redis.del(`user_cases:${session.userId}`);

    // Step 4: Set up real-time updates subscription
    await redis.set(
      `job_user:${ragResponse.jobId}`, 
      session.userId, 
      'EX', 
      3600 // 1 hour
    );

    return json({
      success: true,
      fileId,
      jobId: ragResponse.jobId,
      status: ragResponse.status,
      stages: ragResponse.stages,
    });

  } catch (err) {
    console.error('Evidence upload error:', err);
    throw error(500, 'Upload failed');
  }
};

// Get upload progress/status
export const GET: RequestHandler = async ({ url, cookies }) => {
  const session = await validateSession(cookies);
  if (!session) {
    throw error(401, 'Unauthorized');
  }

  const jobId = url.searchParams.get('jobId');
  if (!jobId) {
    throw error(400, 'Missing jobId');
  }

  try {
    const status = await apiClient.getRAGJobStatus(jobId);
    return json(status);
  } catch (err) {
    console.error('Status check error:', err);
    throw error(500, 'Status check failed');
  }
};
```

```svelte
<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import { invalidate } from '$app/navigation';
  import { Button } from 'bits-ui';
  import { Progress } from 'bits-ui';
  import { AlertCircle, Upload, CheckCircle, FileText } from 'lucide-svelte';
  
  interface Props {
    caseId: string;
    maxFileSize?: number;
    allowedTypes?: string[];
    processingOptions?: {
      generateSummary?: boolean;
      extractCitations?: boolean;
      performEntityExtraction?: boolean;
    };
  }

  let {
    caseId,
    maxFileSize = 50 * 1024 * 1024, // 50MB
    allowedTypes = ['.pdf', '.docx', '.txt', '.png', '.jpg'],
    processingOptions = {
      generateSummary: true,
      extractCitations: true,
      performEntityExtraction: true,
    }
  }: Props = $props();

  const dispatch = createEventDispatcher();

  let uploadState = $state<'idle' | 'uploading' | 'processing' | 'completed' | 'error'>('idle');
  let uploadProgress = $state(0);
  let processingStages = $state<Array<{ name: string; status: string; progress: number }>>([]);
  let errorMessage = $state('');
  let dragOver = $state(false);
  let currentJobId = $state<string | null>(null);
  
  let fileInput: HTMLInputElement;

  const handleFileSelect = (files: FileList | null) => {
    if (!files || files.length === 0) return;
    
    const file = files[0];
    
    // Validate file
    if (file.size > maxFileSize) {
      errorMessage = 'File size exceeds maximum allowed size';
      uploadState = 'error';
      return;
    }

    const fileExtension = '.' + file.name.split('.').pop()?.toLowerCase();
    if (!allowedTypes.includes(fileExtension)) {
      errorMessage = 'File type not supported';
      uploadState = 'error';
      return;
    }

    uploadFile(file);
  };

  const uploadFile = async (file: File) => {
    uploadState = 'uploading';
    errorMessage = '';
    uploadProgress = 0;

    try {
      const formData = new FormData();
      formData.append('file', file);
      formData.append('caseId', caseId);
      formData.append('options', JSON.stringify(processingOptions));

      const response = await fetch('/api/evidence/upload', {
        method: 'POST',
        body: formData,
      });

      if (!response.ok) {
        throw new Error(`Upload failed: ${response.statusText}`);
      }

      const result = await response.json();
      
      if (result.success) {
        currentJobId = result.jobId;
        uploadState = 'processing';
        processingStages = result.stages;
        
        // Start polling for processing status
        pollProcessingStatus(result.jobId);
        
        dispatch('uploadStarted', {
          fileId: result.fileId,
          jobId: result.jobId,
          filename: file.name
        });
      }

    } catch (error) {
      console.error('Upload error:', error);
      errorMessage = error instanceof Error ? error.message : 'Upload failed';
      uploadState = 'error';
    }
  };

  const pollProcessingStatus = async (jobId: string) => {
    try {
      const response = await fetch(`/api/evidence/upload?jobId=${jobId}`);
      const status = await response.json();
      
      processingStages = status.stages;
      
      if (status.status === 'completed') {
        uploadState = 'completed';
        
        // Invalidate case data to refresh the UI
        await invalidate(`/cases/${caseId}`);
        
        dispatch('uploadCompleted', {
          jobId,
          status
        });
        
      } else if (status.status === 'failed') {
        uploadState = 'error';
        errorMessage = 'Processing failed';
        
      } else {
        // Continue polling
        setTimeout(() => pollProcessingStatus(jobId), 2000);
      }
      
    } catch (error) {
      console.error('Status polling error:', error);
      uploadState = 'error';
      errorMessage = 'Failed to check processing status';
    }
  };

  const handleDragOver = (e: DragEvent) => {
    e.preventDefault();
    dragOver = true;
  };

  const handleDragLeave = (e: DragEvent) => {
    e.preventDefault();
    dragOver = false;
  };

  const handleDrop = (e: DragEvent) => {
    e.preventDefault();
    dragOver = false;
    handleFileSelect(e.dataTransfer?.files);
  };

  const resetUploader = () => {
    uploadState = 'idle';
    uploadProgress = 0;
    processingStages = [];
    errorMessage = '';
    currentJobId = null;
  };
</script>
```

```svelte
<script lang="ts">
  import { createEventDispatcher } from 'svelte';
  import { Command, CommandInput, CommandList, CommandEmpty, CommandGroup, CommandItem } from 'bits-ui';
  import { Search, File, Plus, Download, Settings } from 'lucide-svelte';
  
  interface Props {
    open?: boolean;
    caseId?: string;
  }

  let { open = $bindable(), caseId }: Props = $props();
  
  const dispatch = createEventDispatcher();
  
  let searchValue = $state('');
  let searchResults = $state<Array<{ id: string; title: string; type: string; score?: number }>>([]);
  let isSearching = $state(false);

  // Debounced search
  let searchTimeout: NodeJS.Timeout;
  
  const performSearch = async (query: string) => {
    if (!query.trim() || !caseId) return;
    
    isSearching = true;
    
    try {
      const response = await fetch('/api/search', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ 
          query, 
          caseId,
          filters: {} 
        }),
      });
      
      const data = await response.json();
      searchResults = data.results || [];
      
    } catch (error) {
      console.error('Search error:', error);
      searchResults = [];
    } finally {
      isSearching = false;
    }
  };

  const handleSearch = (value: string) => {
    searchValue = value;
    
    clearTimeout(searchTimeout);
    searchTimeout = setTimeout(() => {
      performSearch(value);
    }, 300);
  };

  const handleSelectItem = (item: any) => {
    dispatch('selectItem', item);
    open = false;
    searchValue = '';
    searchResults = [];
  };

  const commands = [
    { id: 'upload', title: 'Upload Evidence', icon: Plus, shortcut: 'Cmd+U' },
    { id: 'search', title: 'Search Documents', icon: Search, shortcut: 'Cmd+K' },
    { id: 'export', title: 'Generate Report', icon: Download, shortcut: 'Cmd+E' },
    { id: 'settings', title: 'Case Settings', icon: Settings, shortcut: 'Cmd+,' },
  ];

  // Keyboard shortcuts
  let handleKeydown = (e: KeyboardEvent) => {
    if (e.metaKey || e.ctrlKey) {
      switch (e.key) {
        case 'k':
          e.preventDefault();
          open = true;
          break;
        case 'u':
          e.preventDefault();
          if (caseId) {
            dispatch('selectItem', { id: 'upload', type: 'command' });
            open = false;
          }
          break;
        case 'e':
          e.preventDefault();
          if (caseId) {
            dispatch('selectItem', { id: 'export', type: 'command' });
            open = false;
          }
          break;
      }
    }
    
    if (e.key === 'Escape') {
      open = false;
    }
  };

  // Mount keyboard listeners
  $effect(() => {
    document.addEventListener('keydown', handleKeydown);
    return () => document.removeEventListener('keydown', handleKeydown);
  });
</script>
```

```typescript
export const load: PageServerLoad = async ({ params, cookies, depends, url }) => {
  // Set up cache dependencies
  depends(`case:${params.caseId}`);
  depends(`case:${params.caseId}:documents`);
  
  const session = await validateSession(cookies);
  if (!session) {
    throw redirect(302, '/login');
  }

  const caseId = params.caseId;

  try {
    // Check cache for case data
    const cacheKey = `case_detail:${caseId}`;
    let caseData = await redis.get(cacheKey);
    
    if (!caseData) {
      // Fetch from API if not cached
      const response = await fetch(`${process.env.API_BASE_URL}/cases/${caseId}`, {
        headers: {
          'Authorization': `Bearer ${session.token}`,
        },
      });
      
      if (!response.ok) {
        throw error(404, 'Case not found');
      }
      
      caseData = await response.text();
      
      // Cache for 5 minutes
      await redis.setex(cacheKey, 300, caseData);
    }

    const case_ = JSON.parse(caseData);

    // Load documents with caching
    const documentsKey = `case_documents:${caseId}`;
    let documentsData = await redis.get(documentsKey);
    
    if (!documentsData) {
      const documentsResponse = await fetch(`${process.env.API_BASE_URL}/cases/${caseId}/documents`, {
        headers: {
          'Authorization': `Bearer ${session.token}`,
        },
      });
      
      documentsData = await documentsResponse.text();
      await redis.setex(documentsKey, 180, documentsData); // 3 minutes
    }

    const documents = JSON.parse(documentsData);

    // Check for any active processing jobs
    const activeJobs = await getActiveProcessingJobs(caseId);

    return {
      case: case_,
      documents,
      activeJobs,
      user: session.user,
    };

  } catch (err) {
    console.error('Failed to load case:', err);
    throw error(500, 'Failed to load case data');
  }
};

export const actions: Actions = {
  // Upload evidence with cache invalidation
  uploadEvidence: async ({ request, params, cookies }) => {
    const session = await validateSession(cookies);
    if (!session) {
      throw error(401, 'Unauthorized');
    }

    const formData = await request.formData();
    const file = formData.get('file') as File;

    try {
      // Process upload via API client
      const result = await apiClient.uploadFile(file, params.caseId);
      
      // Trigger RAG processing
      await apiClient.triggerRAGIngestion({
        fileId: result.fileId,
        caseId: params.caseId,
        userId: session.userId,
        processingOptions: {
          generateSummary: true,
          extractCitations: true,
          performEntityExtraction: true,
          enableSemanticSearch: true,
        },
      });

      // Invalidate relevant caches
      await redis.del(`case_documents:${params.caseId}`);
      await redis.del(`case_detail:${params.caseId}`);
      await redis.del(`user_cases:${session.userId}`);

      return {
        success: true,
        fileId: result.fileId,
        message: 'Evidence uploaded and processing started',
      };

    } catch (err) {
      console.error('Upload action error:', err);
      return {
        success: false,
        error: 'Upload failed',
      };
    }
  },

  // Generate report with caching
  generateReport: async ({ request, params, cookies }) => {
    const session = await validateSession(cookies);
    if (!session) {
      throw error(401, 'Unauthorized');
    }

    const formData = await request.formData();
    const options = {
      includeAllDocuments: formData.get('includeAll') === 'true',
      documentIds: formData.getAll('documentIds') as string[],
      template: formData.get('template') as string || 'default',
      format: formData.get('format') as 'pdf' | 'docx' | 'html' || 'pdf',
    };

    try {
      const result = await apiClient.generateReport(params.caseId, options);
      
      // Cache report URL for download
      await redis.setex(
        `report:${result.reportId}`, 
        3600, // 1 hour
        JSON.stringify({ 
          url: result.downloadUrl, 
          generated: new Date().toISOString() 
        })
      );

      return {
        success: true,
        reportId: result.reportId,
        downloadUrl: result.downloadUrl,
      };

    } catch (err) {
      console.error('Report generation error:', err);
      return {
        success: false,
        error: 'Report generation failed',
      };
    }
  },
};

async function getActiveProcessingJobs(caseId: string) {
  // Get all job keys for this case
  const jobKeys = await redis.keys(`job_case:${caseId}:*`);
  
  if (jobKeys.length === 0) return [];

  // Get job statuses
  const jobStatuses = await Promise.all(
    jobKeys.map(async (key) => {
      const jobId = key.split(':')[2];
      const status = await redis.get(`job_status:${jobId}`);
      return status ? { jobId, ...JSON.parse(status) } : null;
    })
  );

  return jobStatuses.filter(job => 
    job && ['processing', 'queued'].includes(job.status)
  );
}
```

## 5. Real-time Updates with Server-Sent Events

```typescript
// src/routes/api/events/+server.ts
import type { RequestHandler } from './$types';
import { validateSession } from '$lib/server/auth';
import { redis } from '$lib/server/redis';

export const GET: RequestHandler = async ({ url, cookies }) => {
  const session = await validateSession(cookies);
  if (!session) {
    return new Response('Unauthorized', { status: 401 });
  }

  const caseId = url.searchParams.get('caseId');
  if (!caseId) {
    return new Response('Missing caseId', { status: 400 });
  }

  // Set up Server-Sent Events
  const stream = new ReadableStream({
    start(controller) {
      const encoder = new TextEncoder();
      
      // Send initial connection message
      controller.enqueue(encoder.encode(`data: ${JSON.stringify({
        type: 'connected',
        timestamp: new Date().toISOString()
      })}\n\n`));

      // Set up Redis subscription for real-time updates
      const subscriber = redis.duplicate();
      
      subscriber.subscribe(`case:${caseId}:updates`, (err, count) => {
        if (err) {
          console.error('Redis subscription error:', err);
          return;
        }
        console.log(`Subscribed to ${count} channels`);
      });

      subscriber.on('message', (channel, message) => {
        try {
          const update = JSON.parse(message);
          const sseData = `data: ${JSON.stringify(update)}\n\n`;
          controller.enqueue(encoder.encode(sseData));
        } catch (error) {
          console.error('Error processing SSE message:', error);
        }
      });

      // Heartbeat to keep connection alive
      const heartbeat = setInterval(() => {
        controller.enqueue(encoder.encode(`data: ${JSON.stringify({
          type: 'heartbeat',
          timestamp: new Date().toISOString()
        })}\n\n`));
      }, 30000); // 30 seconds

      // Cleanup on close
      return () => {
        clearInterval(heartbeat);
        subscriber.unsubscribe();
        subscriber.quit();
      };
    }
  });

  return new Response(stream, {
    headers: {
      'Content-Type': 'text/event-stream',
      'Cache-Control': 'no-cache',
      'Connection': 'keep-alive',
      'Access-Control-Allow-Origin': '*',
      'Access-Control-Allow-Headers': 'Cache-Control',
    },
  });
};
```
