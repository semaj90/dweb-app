# Production Deployment & Scaling Guide
# Hybrid TypeScript + Go Microservices Architecture

## Complete Development Workflow

### Local Development Environment Setup
```bash
# Project structure
ai-platform/
├── sveltekit-frontend/           # SvelteKit 2 + TypeScript
├── services/
│   ├── data-api-service/         # TypeScript + Drizzle ORM
│   ├── vector-api-service/       # TypeScript + AI/ML
│   ├── storage-service/          # Go + MinIO
│   ├── auth-service/             # Go + JWT
│   └── notification-service/     # Go + WebSocket
├── shared/
│   ├── schema.ts                # Drizzle schema (single source of truth)
│   ├── types/                   # Shared TypeScript types
│   └── proto/                   # gRPC definitions
├── infrastructure/
│   ├── docker-compose.dev.yml   # Development stack
│   ├── k8s/                     # Kubernetes manifests
│   └── terraform/               # Infrastructure as code
└── scripts/
    ├── dev-setup.sh             # Development environment setup
    ├── migrate.ts               # Database migrations
    └── build-all.sh             # Build all services
```

### Development Environment Setup Script
```bash
#!/bin/bash
# scripts/dev-setup.sh

echo "🚀 Setting up AI Platform development environment..."

# Start infrastructure services
echo "📦 Starting PostgreSQL, Redis, Qdrant, MinIO..."
docker-compose -f infrastructure/docker-compose.dev.yml up -d

# Wait for services to be ready
echo "⏳ Waiting for services to be ready..."
sleep 10

# Install dependencies
echo "📚 Installing dependencies..."

# SvelteKit frontend
cd sveltekit-frontend && npm install && cd ..

# TypeScript services
cd services/data-api-service && npm install && cd ../..
cd services/vector-api-service && npm install && cd ../..

# Go services
cd services/storage-service && go mod download && cd ../..
cd services/auth-service && go mod download && cd ../..
cd services/notification-service && go mod download && cd ../..

# Run database migrations
echo "🗄️ Running database migrations..."
cd services/data-api-service && npm run migrate && cd ../..

# Seed development data
echo "🌱 Seeding development data..."
cd services/data-api-service && npm run seed && cd ../..

# Build Go services
echo "🔨 Building Go services..."
cd services/storage-service && go build -o bin/storage-service ./cmd && cd ../..
cd services/auth-service && go build -o bin/auth-service ./cmd && cd ../..
cd services/notification-service && go build -o bin/notification-service ./cmd && cd ../..

echo "✅ Development environment ready!"
echo "🌐 Access the application at http://localhost:3000"
echo "📊 MinIO Console: http://localhost:9001"
echo "🔍 Qdrant Dashboard: http://localhost:6333/dashboard"
```

## Production Infrastructure with Terraform

### AWS Infrastructure Setup
```hcl
# infrastructure/terraform/main.tf

terraform {
  required_version = ">= 1.0"
  required_providers {
    aws = {
      source  = "hashicorp/aws"
      version = "~> 5.0"
    }
  }
}

provider "aws" {
  region = var.aws_region
}

# VPC and Networking
resource "aws_vpc" "main" {
  cidr_block           = "10.0.0.0/16"
  enable_dns_hostnames = true
  enable_dns_support   = true

  tags = {
    Name = "ai-platform-vpc"
  }
}

resource "aws_subnet" "private" {
  count             = 2
  vpc_id            = aws_vpc.main.id
  cidr_block        = "10.0.${count.index + 1}.0/24"
  availability_zone = data.aws_availability_zones.available.names[count.index]

  tags = {
    Name = "ai-platform-private-${count.index + 1}"
  }
}

resource "aws_subnet" "public" {
  count                   = 2
  vpc_id                  = aws_vpc.main.id
  cidr_block              = "10.0.${count.index + 10}.0/24"
  availability_zone       = data.aws_availability_zones.available.names[count.index]
  map_public_ip_on_launch = true

  tags = {
    Name = "ai-platform-public-${count.index + 1}"
  }
}

# EKS Cluster
resource "aws_eks_cluster" "main" {
  name     = "ai-platform-cluster"
  role_arn = aws_iam_role.eks_cluster.arn
  version  = "1.28"

  vpc_config {
    subnet_ids              = concat(aws_subnet.private[*].id, aws_subnet.public[*].id)
    endpoint_private_access = true
    endpoint_public_access  = true
  }

  depends_on = [
    aws_iam_role_policy_attachment.eks_cluster_policy,
    aws_iam_role_policy_attachment.eks_service_policy,
  ]
}

resource "aws_eks_node_group" "main" {
  cluster_name    = aws_eks_cluster.main.name
  node_group_name = "ai-platform-nodes"
  node_role_arn   = aws_iam_role.eks_node.arn
  subnet_ids      = aws_subnet.private[*].id

  instance_types = ["t3.medium", "t3.large"]
  
  scaling_config {
    desired_size = 3
    max_size     = 10
    min_size     = 1
  }

  update_config {
    max_unavailable = 1
  }

  depends_on = [
    aws_iam_role_policy_attachment.eks_worker_node_policy,
    aws_iam_role_policy_attachment.eks_cni_policy,
    aws_iam_role_policy_attachment.eks_container_registry_policy,
  ]
}

# RDS PostgreSQL with pgvector
resource "aws_db_instance" "main" {
  identifier             = "ai-platform-postgres"
  engine                 = "postgres"
  engine_version         = "15.4"
  instance_class         = "db.t3.medium"
  allocated_storage      = 100
  max_allocated_storage  = 1000
  
  db_name  = "ai_platform"
  username = var.db_username
  password = var.db_password
  
  vpc_security_group_ids = [aws_security_group.rds.id]
  db_subnet_group_name   = aws_db_subnet_group.main.name
  
  backup_retention_period = 7
  backup_window          = "03:00-04:00"
  maintenance_window     = "Sun:04:00-Sun:05:00"
  
  skip_final_snapshot = false
  final_snapshot_identifier = "ai-platform-final-snapshot-${formatdate("YYYY-MM-DD-hhmm", timestamp())}"

  tags = {
    Name = "ai-platform-postgres"
  }
}

# ElastiCache Redis Cluster
resource "aws_elasticache_replication_group" "main" {
  replication_group_id       = "ai-platform-redis"

  node_type                  = "cache.t3.micro"

  transit_encryption_enabled = true

  tags = {
    Name = "ai-platform-redis"
  }
}

# S3 Buckets
resource "aws_s3_bucket" "storage" {
  bucket = "${var.project_name}-storage-${random_string.bucket_suffix.result}"

  bucket = aws_s3_bucket.storage.id

    status = "Enabled"

      sse_algorithm = "AES256"

# Application Load Balancer
resource "aws_lb" "main" {
  name               = "ai-platform-alb"
  internal           = false
  load_balancer_type = "application"
  security_groups    = [aws_security_group.alb.id]
  subnets            = aws_subnet.public[*].id

  enable_deletion_protection = false

  tags = {
    Name = "ai-platform-alb"
  }
}

resource "aws_lb_target_group" "sveltekit" {
  name     = "ai-platform-sveltekit"
  port     = 3000
  protocol = "HTTP"
  vpc_id   = aws_vpc.main.id

  health_check {
    enabled             = true
    healthy_threshold   = 2
    interval            = 30
    matcher             = "200"
    path                = "/health"
    port                = "traffic-port"
    protocol            = "HTTP"
    timeout             = 5
    unhealthy_threshold = 2
  }
}

resource "aws_lb_listener" "main" {
  load_balancer_arn = aws_lb.main.arn
  port              = "443"
  protocol          = "HTTPS"
  ssl_policy        = "ELBSecurityPolicy-TLS-1-2-2017-01"
  certificate_arn   = var.ssl_certificate_arn

  default_action {
    type             = "forward"
    target_group_arn = aws_lb_target_group.sveltekit.arn
  }
}

# Security Groups
resource "aws_security_group" "alb" {

  ingress {
    from_port   = 80
```

```typescript
  labelNames: ['time_window'],

// Export all metrics
export const metrics = register;

// Middleware for Express.js services
export function createMetricsMiddleware(serviceName: string) {
  return (req: any, res: any, next: any) => {
    const start = Date.now();
    
    res.on('finish', () => {
      const duration = (Date.now() - start) / 1000;
      const route = req.route?.path || req.url;
      
      httpRequestsTotal.labels({
        method: req.method,
        route,
        status_code: res.statusCode.toString(),
        service: serviceName,
      }).inc();
      
      httpRequestDuration.labels({
        method: req.method,
        route,
        service: serviceName,
      }).observe(duration);
    });
    
    next();
  };
}
```

```yaml
# .github/workflows/deploy-production.yml
name: Deploy to Production

on:
  push:
    branches: [main]
  workflow_dispatch:

env:
  REGISTRY: ghcr.io
  IMAGE_NAME: ${{ github.repository }}

jobs:
  changes:
    runs-on: ubuntu-latest
    outputs:
      sveltekit: ${{ steps.filter.outputs.sveltekit }}
      data-api: ${{ steps.filter.outputs.data-api }}
      vector-api: ${{ steps.filter.outputs.vector-api }}
      storage-service: ${{ steps.filter.outputs.storage-service }}
      auth-service: ${{ steps.filter.outputs.auth-service }}
    steps:
    - uses: actions/checkout@v4
    - uses: dorny/paths-filter@v2
      id: filter
      with:
        filters: |
          sveltekit:
            - 'sveltekit-frontend/**'
            - 'shared/**'
          data-api:
            - 'services/data-api-service/**'
            - 'shared/**'
          vector-api:
            - 'services/vector-api-service/**'
            - 'shared/**'
          storage-service:
            - 'services/storage-service/**'
          auth-service:
            - 'services/auth-service/**'

  test-services:
    needs: changes
    runs-on: ubuntu-latest
    services:
      postgres:
        image: pgvector/pgvector:pg15
        env:
          POSTGRES_PASSWORD: postgres
          POSTGRES_DB: test_db
        options: >-
          --health-cmd pg_isready
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 5432:5432
      redis:
        image: redis:7
        options: >-
          --health-cmd "redis-cli ping"
          --health-interval 10s
          --health-timeout 5s
          --health-retries 5
        ports:
          - 6379:6379

    strategy:
      matrix:
        service: [sveltekit, data-api, vector-api, storage-service, auth-service]
        include:
          - service: sveltekit
            condition: needs.changes.outputs.sveltekit == 'true'
            path: sveltekit-frontend
            runtime: node
          - service: data-api
            condition: needs.changes.outputs.data-api == 'true'
            path: services/data-api-service
            runtime: node
          - service: vector-api
            condition: needs.changes.outputs.vector-api == 'true'
            path: services/vector-api-service
            runtime: node
          - service: storage-service
            condition: needs.changes.outputs.storage-service == 'true'
            path: services/storage-service
            runtime: go
          - service: auth-service
            condition: needs.changes.outputs.auth-service == 'true'
            path: services/auth-service
            runtime: go

    steps:
    - uses: actions/checkout@v4
      if: ${{ matrix.condition }}

    - name: Setup Node.js
      if: ${{ matrix.condition && matrix.runtime == 'node' }}
      uses: actions/setup-node@v4
      with:
        node-version: '20'
        cache: 'npm'
        cache-dependency-path: ${{ matrix.path }}/package-lock.json

    - name: Setup Go
      if: ${{ matrix.condition && matrix.runtime == 'go' }}
      uses: actions/setup-go@v4
      with:
        go-version: '1.21'

    - name: Install Node.js dependencies
      if: ${{ matrix.condition && matrix.runtime == 'node' }}
      run: cd ${{ matrix.path }} && npm ci

    - name: Install Go dependencies
      if: ${{ matrix.condition && matrix.runtime == 'go' }}
      run: cd ${{ matrix.path }} && go mod download

    - name: Run Node.js tests
      if: ${{ matrix.condition && matrix.runtime == 'node' }}
      run: cd ${{ matrix.path }} && npm run test
      env:
        DATABASE_URL: postgres://postgres:postgres@localhost:5432/test_db?sslmode=disable
        REDIS_URL: redis://localhost:6379

    - name: Run Go tests
      if: ${{ matrix.condition && matrix.runtime == 'go' }}
      run: cd ${{ matrix.path }} && go test -v ./...
      env:
        DATABASE_URL: postgres://postgres:postgres@localhost:5432/test_db?sslmode=disable
        REDIS_URL: redis://localhost:6379

  build-and-push:
    needs: [changes, test-services]
    runs-on: ubuntu-latest
    if: always() && needs.test-services.result == 'success'

    strategy:
      matrix:
        service: [sveltekit, data-api, vector-api, storage-service, auth-service]
        include:
          - service: sveltekit
            condition: needs.changes.outputs.sveltekit == 'true'
            path: sveltekit-frontend
            dockerfile: sveltekit-frontend/Dockerfile
          - service: data-api
            condition: needs.changes.outputs.data-api == 'true'
            path: services/data-api-service
            dockerfile: services/data-api-service/Dockerfile
          - service: vector-api
            condition: needs.changes.outputs.vector-api == 'true'
            path: services/vector-api-service
            dockerfile: services/vector-api-service/Dockerfile
          - service: storage-service
            condition: needs.changes.outputs.storage-service == 'true'
            path: services/storage-service
            dockerfile: services/storage-service/Dockerfile
          - service: auth-service
            condition: needs.changes.outputs.auth-service == 'true'
            path: services/auth-service
            dockerfile: services/auth-service/Dockerfile

    steps:
    - uses: actions/checkout@v4
      if: ${{ matrix.condition }}

    - name: Log in to Container Registry
      if: ${{ matrix.condition }}
      uses: docker/login-action@v3
      with:
        registry: ${{ env.REGISTRY }}
        username: ${{ github.actor }}
        password: ${{ secrets.GITHUB_TOKEN }}

    - name: Extract metadata
      if: ${{ matrix.condition }}
      id: meta
      uses: docker/metadata-action@v5
      with:
        images: ${{ env.REGISTRY }}/${{ env.IMAGE_NAME }}/${{ matrix.service }}
        tags: |
          type=ref,event=branch
          type=ref,event=pr
          type=sha,prefix={{branch}}-
          type=raw,value=latest,enable={{is_default_branch}}

    - name: Build and push Docker image
      if: ${{ matrix.condition }}
      uses: docker/build-push-action@v5
      with:
        context: .
        file: ${{ matrix.dockerfile }}
        push: true
        tags: ${{ steps.meta.outputs.tags }}
        labels: ${{ steps.meta.outputs.labels }}

  deploy:
    needs: [changes, build-and-push]
    runs-on: ubuntu-latest
    if: always() && needs.build-and-push.result == 'success'

    steps:
    - uses: actions/checkout@v4

    - name: Configure kubectl
      uses: azure/k8s-set-context@v3
      with:
        method: kubeconfig
        kubeconfig: ${{ secrets.KUBE_CONFIG }}

    - name: Update image tags in manifests
      run: |
        # Update image tags to use the commit SHA
        sed -i 's|:latest|:main-${{ github.sha }}|g' infrastructure/k8s/*.yaml

    - name: Deploy to Kubernetes
      run: |
        kubectl apply -f infrastructure/k8s/
        
        # Wait for deployments to be ready
        kubectl rollout status deployment/sveltekit-app -n ai-platform --timeout=600s
        kubectl rollout status deployment/data-api-service -n ai-platform --timeout=600s
        kubectl rollout status deployment/vector-api-service -n ai-platform --timeout=600s
        kubectl rollout status deployment/storage-service -n ai-platform --timeout=600s
        kubectl rollout status deployment/auth-service -n ai-platform --timeout=600s

    - name: Run smoke tests
      run: |
        # Wait a bit for services to be fully ready
        sleep 30
        
        # Get the ingress IP
        INGRESS_IP=$(kubectl get ingress ai-platform-ingress -n ai-platform -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
        
        # Test health endpoints
        curl -f http://$INGRESS_IP/health || exit 1
        curl -f http://$INGRESS_IP/api/data/health || exit 1
        curl -f http://$INGRESS_IP/api/vector/health || exit 1

    - name: Notify deployment success
      if: success()
      run: |
        echo "✅ Deployment successful!"
        echo "🌐 Application URL: https://your-domain.com"
        echo "📊 Monitoring: https://grafana.your-domain.com"

    - name: Notify deployment failure
      if: failure()
      run: |
        echo "❌ Deployment failed!"
        kubectl get pods -n ai-platform
        kubectl describe pods -n ai-platform
```
